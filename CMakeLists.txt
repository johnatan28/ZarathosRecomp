cmake_minimum_required(VERSION 3.21)

project(ghostcomp LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

set(GHOSTCOMP_PS2RECOMP_DIR "${CMAKE_SOURCE_DIR}/tools/PS2Recomp"
    CACHE PATH "Path to the PS2Recomp checkout")
set(GHOSTCOMP_PS2RECOMP_BUILD_DIR "${GHOSTCOMP_PS2RECOMP_DIR}/b"
    CACHE PATH "Path to the prebuilt PS2Recomp CMake build directory")
set(GHOSTCOMP_GENERATED_DIR "${CMAKE_SOURCE_DIR}/generated"
    CACHE PATH "Path to the local C++ files generated by ps2_recomp")
set(GHOSTCOMP_DEFAULT_ELF "${CMAKE_SOURCE_DIR}/game/SLUS_213.06"
    CACHE FILEPATH "Default path to user-supplied Ghost Rider (PS2) ELF")

option(GHOSTCOMP_UNITY_BUILD "Enable CMake Unity Build for fast compilation of recompiled functions" ON)
set(GHOSTCOMP_UNITY_BATCH_SIZE 64 CACHE STRING "Number of generated .cpp files per unity batch")
option(GHOSTCOMP_USE_PCH "Enable precompiled headers for generated files" ON)

if(NOT EXISTS "${GHOSTCOMP_GENERATED_DIR}/register_functions.cpp")
    message(FATAL_ERROR
        "Generated C++ sources not found under '${GHOSTCOMP_GENERATED_DIR}'.\n"
        "Run ps2_recomp first:\n"
        "  ps2_recomp re/ghostrider.toml")
endif()

if(NOT EXISTS "${GHOSTCOMP_PS2RECOMP_BUILD_DIR}/ps2xRuntime/Release/ps2_runtime.lib" AND WIN32)
    message(FATAL_ERROR
        "Prebuilt PS2Recomp Release libraries not found under '${GHOSTCOMP_PS2RECOMP_BUILD_DIR}'.\n"
        "Build PS2Recomp in Release mode first.")
endif()

file(GLOB GHOSTCOMP_GENERATED_SOURCES CONFIGURE_DEPENDS
    "${GHOSTCOMP_GENERATED_DIR}/*.cpp"
)

add_executable(ghostcomp
    runtime/src/main.cpp
    runtime/src/ghostrider_overrides.cpp
    ${GHOSTCOMP_GENERATED_SOURCES}
)

target_include_directories(ghostcomp PRIVATE
    "${GHOSTCOMP_GENERATED_DIR}"
    "${GHOSTCOMP_PS2RECOMP_DIR}/ps2xRuntime/include"
    "${GHOSTCOMP_PS2RECOMP_DIR}/ps2xRuntime/src/lib/Kernel"
    "${GHOSTCOMP_PS2RECOMP_DIR}/ps2xIOP/include"
    "${GHOSTCOMP_PS2RECOMP_BUILD_DIR}/_deps/raylib-src/src"
)

set(GHOSTCOMP_DEFAULT_ELF_ESCAPED "${GHOSTCOMP_DEFAULT_ELF}")
string(REPLACE "\\" "\\\\" GHOSTCOMP_DEFAULT_ELF_ESCAPED "${GHOSTCOMP_DEFAULT_ELF_ESCAPED}")
string(REPLACE "\"" "\\\"" GHOSTCOMP_DEFAULT_ELF_ESCAPED "${GHOSTCOMP_DEFAULT_ELF_ESCAPED}")

target_compile_definitions(ghostcomp PRIVATE
    GHOSTCOMP_DEFAULT_ELF="${GHOSTCOMP_DEFAULT_ELF_ESCAPED}"
    PS2_RUNTIME_LOGS=1
)

if(GHOSTCOMP_UNITY_BUILD)
    set_target_properties(ghostcomp PROPERTIES
        UNITY_BUILD ON
        UNITY_BUILD_BATCH_SIZE "${GHOSTCOMP_UNITY_BATCH_SIZE}"
    )
    set_source_files_properties(
        runtime/src/main.cpp
        runtime/src/ghostrider_overrides.cpp
        "${GHOSTCOMP_GENERATED_DIR}/register_functions.cpp"
        PROPERTIES SKIP_UNITY_BUILD_INCLUSION TRUE
    )
endif()

if(GHOSTCOMP_USE_PCH)
    target_precompile_headers(ghostcomp PRIVATE
        "${GHOSTCOMP_PS2RECOMP_DIR}/ps2xRuntime/include/ps2_runtime_macros.h"
        "${GHOSTCOMP_PS2RECOMP_DIR}/ps2xRuntime/include/ps2_runtime.h"
        "${GHOSTCOMP_GENERATED_DIR}/ps2_recompiled_functions.h"
        "${GHOSTCOMP_GENERATED_DIR}/ps2_recompiled_stubs.h"
        "${GHOSTCOMP_PS2RECOMP_DIR}/ps2xRuntime/include/ps2_syscalls.h"
        "${GHOSTCOMP_PS2RECOMP_DIR}/ps2xRuntime/include/ps2_stubs.h"
    )
endif()

if(MSVC)
    target_compile_options(ghostcomp PRIVATE /FS /bigobj /wd4101 /wd4102 /wd4244 /wd4267 /wd4305 /wd4805)
    # Same workaround used by PS2Recomp's ps2EntryRunner for WinAPI vs raylib CloseWindow/ShowCursor clash
    target_link_options(ghostcomp PRIVATE "/FORCE:MULTIPLE")
endif()

set(PS2X_FFMPEG_BIN_DIR "${GHOSTCOMP_PS2RECOMP_BUILD_DIR}/ThirdParty/ffmpeg-prefix/src/ffmpeg_external/bin")

target_link_libraries(ghostcomp PRIVATE
    "${GHOSTCOMP_PS2RECOMP_BUILD_DIR}/ps2xRuntime/Release/ps2_runtime.lib"
    "${GHOSTCOMP_PS2RECOMP_BUILD_DIR}/ps2xIOP/Release/ps2_iop.lib"
    "${GHOSTCOMP_PS2RECOMP_BUILD_DIR}/_deps/raylib-build/raylib/Release/raylib.lib"
    "${PS2X_FFMPEG_BIN_DIR}/avcodec.lib"
    "${PS2X_FFMPEG_BIN_DIR}/avformat.lib"
    "${PS2X_FFMPEG_BIN_DIR}/avutil.lib"
    "${PS2X_FFMPEG_BIN_DIR}/swresample.lib"
    "${PS2X_FFMPEG_BIN_DIR}/swscale.lib"
)

if(WIN32)
    target_link_libraries(ghostcomp PRIVATE
        bcrypt
        secur32
        ws2_32
        opengl32
        glu32
        winmm
        kernel32
        user32
        gdi32
        winspool
        shell32
        ole32
        oleaut32
        uuid
        comdlg32
        advapi32
    )

    if(EXISTS "${PS2X_FFMPEG_BIN_DIR}")
        add_custom_command(TARGET ghostcomp POST_BUILD
            COMMAND "${CMAKE_COMMAND}" -E copy_directory
            "${PS2X_FFMPEG_BIN_DIR}"
            "$<TARGET_FILE_DIR:ghostcomp>"
            COMMENT "Staging FFmpeg runtime DLLs next to ghostcomp.exe"
        )
    endif()
endif()
